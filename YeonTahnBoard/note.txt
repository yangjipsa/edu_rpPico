from machine import Pin
import time

sw1 = Pin(3, Pin.IN)
sw2 = Pin(4, Pin.IN)

# 메인 루프
try:
    while True:
        valSW1 = sw1.value()
        valSW2 = sw2.value()
        time.sleep(0.01)
        
        print(f"SW1 : {valSW1} | SW2 : {valSW2}")
        
except KeyboardInterrupt :
    print("Turn off")
